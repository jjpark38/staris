2.3 Round 정수 나눗셈 + MythicEssence 음수 보호 (dry-run §11 R-DR-S03 가드)

// PlayerPrestigeState.MigrateSeasonalToPermanent(float retainRatio = 0.5f)
public void MigrateSeasonalToPermanent(float retainRatio) {
  // ... (1~3 순서) ...

  // Step 4: Round 정수 연산 강제 (Q6 결정 = B Round)
  long seasonalTotal = SeasonalPp + Σ slot.investedPp where channel=Seasonal;
  long convertedPermanent = (seasonalTotal * 50L) / 100L;  // (long * 50) / 100 = Round 자연 처리

  // 음수 보호 (R-DR-S03)
  if (convertedPermanent < 0L) {
    Debug.LogError($"[Season] Migrate convertedPermanent < 0: {convertedPermanent}, seasonalTotal={seasonalTotal}");
    convertedPermanent = 0L;  // Fail-safe = 0 로 처리, throw 안 함 (유저 PP 보호 우선)
  }

  PermanentPp += convertedPermanent;
  SeasonalPp = 0L;
  RemoveAll(s => s.channel == Seasonal);
}

// PlayerWalletState.RetainMythicEssenceForNextSeason(float retainRatio = 0.5f)
public void RetainMythicEssenceForNextSeason(float retainRatio) {
  long retained = (MythicEssence * 50L) / 100L;
  if (retained < 0L) {
    Debug.LogError($"[Season] MythicEssence retain < 0: {retained}");
    retained = 0L;
  }
  MythicEssence = retained;
}
